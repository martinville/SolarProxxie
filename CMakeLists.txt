cmake_minimum_required(VERSION 3.16)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
             "${CMAKE_CURRENT_LIST_DIR}/VERSION")
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/VERSION" PROJECT_VER LIMIT_COUNT 1)
string(STRIP "${PROJECT_VER}" PROJECT_VER)
if(NOT PROJECT_VER MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+([+-][0-9A-Za-z.-]+)?$")
    message(FATAL_ERROR "VERSION must contain a semantic version (found '${PROJECT_VER}')")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(SolarProxxie)
if(NOT IDF_TARGET STREQUAL "esp32")
    message(FATAL_ERROR "SolarProxxie 1.0 targets the original dual-core ESP32 only")
endif()
idf_component_get_property(lwip_lib lwip COMPONENT_LIB)
target_include_directories(${lwip_lib} PRIVATE "${PROJECT_DIR}/main/router")
target_compile_definitions(${lwip_lib} PRIVATE ESP_IDF_LWIP_HOOK_FILENAME="ghost_hooks.h")
idf_component_get_property(json_lib json COMPONENT_LIB)
target_compile_definitions(${json_lib} PRIVATE CJSON_NESTING_LIMIT=12)
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${PROJECT_DIR}"
                OUTPUT_VARIABLE ghost_commit OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET
                RESULT_VARIABLE ghost_git_result)
if(NOT ghost_git_result EQUAL 0)
    set(ghost_commit "unavailable")
endif()
idf_component_get_property(main_lib main COMPONENT_LIB)
target_compile_definitions(${main_lib} PRIVATE GHOST_GIT_COMMIT="${ghost_commit}")
